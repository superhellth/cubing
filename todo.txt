const getConsistencyTier = (score) => {
    if (score > 4) return "🤖 Machine"; // < 0.25s SD
    if (score > 2) return "💎 Elite";   // < 0.5s SD
    if (score > 1) return "🔥 Solid";   // < 1.0s SD
    if (score > 0.5) return "⚖️ Okay";  // < 2.0s SD
    return "🎲 Volatile";
};
